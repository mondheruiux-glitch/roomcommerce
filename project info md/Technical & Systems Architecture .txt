Database Architecture & Data Schemas3.1 Relational Schema (PostgreSQL 16)SQL-- Core User Accounts
CREATE TABLE users (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    email VARCHAR(255) UNIQUE NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    free_renders_used SMALLINT NOT NULL DEFAULT 0,
    has_active_pass BOOLEAN NOT NULL DEFAULT FALSE
);

-- Move-In Pass Entitlements (Episodic 30-Day Billing)
CREATE TABLE move_passes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    stripe_session_id VARCHAR(255) UNIQUE NOT NULL,
    amount_cents INTEGER NOT NULL DEFAULT 1499,
    currency VARCHAR(3) NOT NULL DEFAULT 'USD',
    activated_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    expires_at TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '30 days'),
    generations_executed SMALLINT NOT NULL DEFAULT 0,
    is_active BOOLEAN GENERATED ALWAYS AS (
        NOW() < expires_at AND generations_executed < 60
    ) STORED
);
CREATE INDEX idx_move_passes_user ON move_passes(user_id) WHERE NOW() < expires_at;

-- Room Design Projects
CREATE TYPE style_preset_enum AS ENUM (
    'JAPANDI', 'MID_CENTURY_MODERN', 'SCANDINAVIAN', 'INDUSTRIAL',
    'MINIMALIST', 'CONTEMPORARY', 'BOHEMIAN', 'MODERN_FARMHOUSE'
);

CREATE TYPE room_function_enum AS ENUM (
    'LIVING_ROOM', 'BEDROOM', 'HOME_OFFICE', 'DINING_ROOM', 'PATIO'
);

CREATE TABLE projects (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE CASCADE,
    guest_session_id VARCHAR(128),
    style_preset style_preset_enum NOT NULL,
    room_function room_function_enum NOT NULL DEFAULT 'LIVING_ROOM',
    budget_ceiling_cents INTEGER NOT NULL,
    currency VARCHAR(3) NOT NULL DEFAULT 'USD',
    raw_image_url TEXT NOT NULL,
    depth_map_url TEXT,
    rendered_image_url TEXT,
    purge_scheduled_at TIMESTAMPTZ NOT NULL DEFAULT (NOW() + INTERVAL '7 days'),
    status VARCHAR(32) NOT NULL DEFAULT 'PENDING',
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_projects_user ON projects(user_id);
CREATE INDEX idx_projects_purge ON projects(purge_scheduled_at);

-- Catalog Inventory (Aggregated Multi-Retailer Feed)
CREATE TABLE catalog_skus (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    sku VARCHAR(128) NOT NULL,
    retailer_id VARCHAR(32) NOT NULL, -- 'IKEA', 'TARGET', 'WAYFAIR', 'AMAZON'
    title VARCHAR(512) NOT NULL,
    category VARCHAR(64) NOT NULL,
    price_cents INTEGER NOT NULL,
    currency VARCHAR(3) NOT NULL DEFAULT 'USD',
    width_cm NUMERIC(6,2),
    depth_cm NUMERIC(6,2),
    height_cm NUMERIC(6,2),
    canonical_url TEXT NOT NULL,
    affiliate_url TEXT NOT NULL,
    primary_image_url TEXT NOT NULL,
    in_stock BOOLEAN NOT NULL DEFAULT TRUE,
    last_verified_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    CONSTRAINT uq_retailer_sku UNIQUE (retailer_id, sku)
);
CREATE INDEX idx_catalog_retrieval ON catalog_skus(category, in_stock, price_cents);

-- Generated Hotspot Pins on Canvas
CREATE TABLE canvas_hotspots (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    matched_sku_id UUID NOT NULL REFERENCES catalog_skus(id),
    category VARCHAR(64) NOT NULL,
    norm_x NUMERIC(5,4) NOT NULL, -- Normalized 0.0000 to 1.0000
    norm_y NUMERIC(5,4) NOT NULL,
    similarity_score NUMERIC(5,4) NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_canvas_hotspots_project ON canvas_hotspots(project_id);

-- Outbound Affiliate Click Telemetry
CREATE TABLE affiliate_click_events (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES users(id) ON DELETE SET NULL,
    project_id UUID NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
    hotspot_id UUID NOT NULL REFERENCES canvas_hotspots(id),
    sku_id UUID NOT NULL REFERENCES catalog_skus(id),
    retailer_id VARCHAR(32) NOT NULL,
    click_id UUID NOT NULL DEFAULT gen_random_uuid(),
    user_agent TEXT,
    ip_hash VARCHAR(64),
    created_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);
CREATE INDEX idx_affiliate_clicks_created ON affiliate_click_events(created_at);

5. Catalog Ingestion & Inventory Synchronization EngineTo satisfy NFR-07 and BR-02 (zero out-of-stock pins), the backend operates an automated ETL synchronization worker.┌────────────────────────────────────────────────────────────────────────────────────────┐
│                              CATALOG ETL & SYNC TOPOLOGY                               │
└────────────────────────────────────────────────────────────────────────────────────────┘

  [Cron Worker: Hourly / Daily Batch]
         │
         ├── Fetch SKU Feeds via Retailer APIs & Affiliates (Impact, CJ, Amazon PA-API)
         │
         ▼
  [Data Transformation & Normalization Pipeline]
         ├── Deduplicate SKUs across retailers.
         ├── Normalize categories to internal taxonomy.
         ├── Convert dimensions to metric (cm) & imperial (in).
         │
         ▼
  [Stock & Availability Health Check]
         ├── If SKU Status == "OUT_OF_STOCK" OR HTTP Status == 404/410:
         │     ├── Update `catalog_skus.in_stock = FALSE` in PostgreSQL.
         │     └── Execute Qdrant Payload Update: Set `in_stock = false`.
         │
         └── If SKU is New / Updated:
               ├── Generate 768d vector via SigLIP inference worker.
               └── Upsert into Qdrant collection `retail_catalog_v1`.
Automated Fallback Logic for Out-of-Stock SKUsIf a SKU goes out of stock between generation time and user click:When SCR-08A opens, the client executes an optimistic real-time availability check via GET /api/v1/catalog/:sku_id/availability.If marked unavailable, the system transparently swaps the primary display with the second-highest ranked vector match from the candidate pool without breaking the pin.6. Security, Privacy & Compliance Architecture6.1 Ephemeral Storage Lifecycle (GDPR / CCPA Mandate)Zero Indefinite Image Storage: User-uploaded domestic photos and generated renders are assigned a strict 7-day retention horizon (SEC-01, PRV-01).Implementation: Cloudflare R2 / AWS S3 buckets enforce lifecycle rules:XML<LifecycleConfiguration>
    <Rule>
        <ID>PurgeUserUploadsAfter7Days</ID>

        <Expiration><Days>7</Days></Expiration>
    </Rule>
    <Rule>
        <ID>PurgeGeneratedRendersAfter7Days</ID>
        <Filter><Prefix>renders/</Prefix></Filter>
        <Status>Enabled</Status>
        <Expiration><Days>7</Days></Expiration>
    </Rule>
</LifecycleConfiguration>
PostgreSQL Cleanup Daemon: A daily cron worker purges expired records from projects and cascading canvas_hotspots matching purge_scheduled_at <= NOW().6.2 Client-Side & Ingestion Sanitization GuardrailsEXIF Erasure: A WebAssembly/JavaScript module strips all EXIF metadata (GPS, camera serials, timestamps) in the user's browser before transmission.Biometric & Face Obfuscation: The serverless ingestion pipeline runs an OpenCV YuNet face-detection model on every raw upload. Any detected face receives a $31\times31$ kernel Gaussian blur prior to passing to the depth/diffusion models (SEC-02).Encryption Standards: TLS 1.3 in transit with HSTS enforced; AES-256 server-side encryption for all stored object artifacts (SEC-03).7.
